from typing import List, Tuple, Set

def boolean_matrix_multiply(
    M_R: List[List[int]], 
    M_S: List[List[int]]
) -> List[List[int]]:
    # Calcula o produto booleano M_R e M_S de duas matrizes lógicas
    rows = len(M_R)
    cols = len(M_S[0])
    inner_dim = len(M_S)
    
    # Inicializa matriz resultante zerada
    result = [[0 for _ in range(cols)] for _ in range(rows)]
    
    for i in range(rows):
        for j in range(cols):
            val = 0
            for k in range(inner_dim):
                if M_R[i][k] and M_S[k][j]:
                    val = 1
                    break  # Curto-circuito lógico para or booleano
            result[i][j] = val
    return result

def compose_relations(
    R: Set[Tuple[any, any]], 
    S: Set[Tuple[any, any]]
) -> Set[Tuple[any, any]]:
    # Composição relacional exata baseada na definição existencial
    composite = set()
    for a, b1 in R:
        for b2, c in S:
            if b1 == b2:
                composite.add((a, c))
    return composite

def warshall_fecho_transitivo(matriz_adj: list[list[int]]) -> list[list[int]]:
    # Computa o fecho transitivo de uma relacao binaria representada por matriz booleana.
    # Complexidade de Tempo: Theta(n^3)
    # Complexidade de Espaço: Theta(n^2) para a matriz resultante
    n = len(matriz_adj)
    # Clona a matriz para W_k
    w = [linha[:] for linha in matriz_adj]

    # k representa o vértice pivô intermediário
    for k in range(n):
        for i in range(n):
            for j in range(n):
                w[i][j] = w[i][j] or (w[i][k] and w[k][j])
                
    return w
